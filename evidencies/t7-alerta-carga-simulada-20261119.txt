# Alertas del grupo 3 en los nodos Proxmox. monit mira el sistema cada 20 s (un "ciclo").
# "for 3 cycles" = el problema tiene que durar 1 minuto seguido para avisar: así un pico corto no dispara la alerta.
check system $HOST
    if cpu usage > 80% for 3 cycles then exec "/usr/local/sbin/g3-alerta CPU-ALTA"
        else if succeeded for 3 cycles then exec "/usr/local/sbin/g3-alerta CPU-NORMAL"
    if loadavg (1min) per core > 2 for 3 cycles then exec "/usr/local/sbin/g3-alerta CARGA-ALTA"
        else if succeeded for 3 cycles then exec "/usr/local/sbin/g3-alerta CARGA-NORMAL"
    if memory usage > 90% for 3 cycles then exec "/usr/local/sbin/g3-alerta MEMORIA-ALTA"
        else if succeeded for 3 cycles then exec "/usr/local/sbin/g3-alerta MEMORIA-NORMAL"

check filesystem raiz with path /
    if space usage > 85% for 3 cycles then exec "/usr/local/sbin/g3-alerta DISCO-LLENO"

check filesystem nas with path /mnt/pve/truenas-nfs
    if does not exist for 3 cycles then exec "/usr/local/sbin/g3-alerta NAS-NO-MONTADO"

===== /usr/local/sbin/g3-alerta
#!/bin/sh
# Apunta la alerta en /var/log/g3-alertas.log y en el registro del sistema (journal).
L="$(date '+%d/%m/%Y %H:%M:%S') $(hostname) ALERTA $1 | carga $(cut -d' ' -f1-3 /proc/loadavg) | $MONIT_DESCRIPTION"
echo "$L" >> /var/log/g3-alertas.log
logger -t g3-alerta -p user.warning "$L"
